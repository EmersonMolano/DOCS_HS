-- ─── Extensiones ─────────────────────────────────────
-- pgcrypto: permite funciones de hash y cifrado (usamos digest() para contraseñas).
-- uuid-ossp: genera UUIDs, los usamos como salt aleatorio en credenciales.
CREATE EXTENSION IF NOT EXISTS pgcrypto;
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- btree_gist: necesaria para el constraint EXCLUDE en hogar_tarifa.
-- EXCLUDE es como un UNIQUE pero para rangos (ej: evitar dos tarifas
-- activas al mismo tiempo para el mismo hogar).
CREATE EXTENSION IF NOT EXISTS btree_gist;


-- =====================================================
-- BLOQUE 1: SEGURIDAD Y CONFIGURACIÓN
-- =====================================================

-- Guarda las reglas que deben cumplir las contraseñas:
-- longitud mínima, si requiere símbolos, cada cuántos días
-- caduca, y cuántos intentos fallidos bloquean la cuenta.
-- NOTA: min_longitud = 8 para cumplir exactamente con ERF1.1.2
-- del SRS (criterio de aceptación: "Mínimo 8 caracteres").
CREATE TABLE politicas_contrasenas (
    politica_id         SERIAL PRIMARY KEY,
    min_longitud        INTEGER  DEFAULT 8,    -- alineado con SRS ERF1.1.2
    max_longitud        INTEGER  DEFAULT 50,
    requiere_mayusculas BOOLEAN  DEFAULT TRUE,
    requiere_numeros    BOOLEAN  DEFAULT TRUE,
    requiere_simbolos   BOOLEAN  DEFAULT TRUE,
    caducidad_dias      INTEGER  DEFAULT 90,
    bloqueo_intentos    INTEGER  DEFAULT 5,
    bloqueo_minutos     INTEGER  DEFAULT 30
);

-- Tabla de parámetros globales del sistema (clave-valor).
-- Permite cambiar comportamientos sin tocar el código:
-- por ejemplo, cuántas sesiones simultáneas permite un usuario.
CREATE TABLE configuracion_seguridad (
    configuracion_id     SERIAL PRIMARY KEY,
    nombre_configuracion VARCHAR(100) NOT NULL UNIQUE,
    valor_configuracion  VARCHAR(200),
    descripcion          VARCHAR(255),
    fecha_modificacion   TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- Un rol es un grupo de permisos con nombre (ej: "Admin", "Técnico").
-- Los usuarios heredan los permisos del rol que se les asigne.
CREATE TABLE roles (
    rol_id      SERIAL PRIMARY KEY,
    nombre_rol  VARCHAR(50)  NOT NULL UNIQUE,
    descripcion VARCHAR(255)
);

-- Un permiso es una acción concreta que puede hacerse en el sistema
-- (ej: "ver_reportes", "gestionar_dispositivos").
-- tipo_permiso:
--   'funcional' → acciones de pantalla o endpoint
--   'dato'      → operaciones SQL directas (SELECT, INSERT…)
CREATE TABLE permisos (
    permiso_id     SERIAL PRIMARY KEY,
    nombre_permiso VARCHAR(50) NOT NULL UNIQUE,
    tipo_permiso   VARCHAR(20) DEFAULT 'funcional'
        CHECK (tipo_permiso IN ('funcional','dato')),
    descripcion    VARCHAR(255)
);

-- Tabla puente que asocia roles con permisos.
-- Un rol puede tener muchos permisos, y un permiso puede
-- pertenecer a varios roles (relación muchos-a-muchos).
CREATE TABLE rol_permiso (
    rol_id           INTEGER NOT NULL REFERENCES roles(rol_id)        ON DELETE CASCADE,
    permiso_id       INTEGER NOT NULL REFERENCES permisos(permiso_id) ON DELETE CASCADE,
    fecha_asignacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (rol_id, permiso_id)
);


-- =====================================================
-- BLOQUE 2: USUARIOS (separados en 5 tablas)
-- =====================================================

-- Identidad central: solo los datos que identifican al usuario
-- dentro del sistema. Todo lo demás va en tablas separadas.
-- tipo_autenticacion define cómo inicia sesión:
--   'Local'  → usuario y contraseña guardados en esta BD
--   'OAuth2' → inicia sesión con Google, GitHub, etc.
--   'LDAP'   → inicia sesión con el directorio corporativo
CREATE TABLE usuarios (
    usuario_id         SERIAL PRIMARY KEY,
    nombre_usuario     VARCHAR(100) NOT NULL UNIQUE,
    email              VARCHAR(150) NOT NULL UNIQUE,
    tipo_autenticacion VARCHAR(20)  NOT NULL DEFAULT 'Local'
        CHECK (tipo_autenticacion IN ('Local','OAuth2','LDAP')),
    fecha_creacion     TIMESTAMPTZ  DEFAULT CURRENT_TIMESTAMP
);

-- Datos personales del usuario: nombre, documento, teléfono, ciudad.
-- Separados de la identidad para poder aplicar cifrado o
-- restricciones de acceso solo a esta tabla (RLS más fino).
CREATE TABLE perfil_usuario (
    perfil_id        SERIAL PRIMARY KEY,
    usuario_id       INTEGER NOT NULL UNIQUE
        REFERENCES usuarios(usuario_id) ON DELETE CASCADE,
    nombre_completo  VARCHAR(200) NOT NULL,
    tipo_documento   VARCHAR(20),
    numero_documento VARCHAR(50) UNIQUE,
    telefono         VARCHAR(20),
    ciudad           VARCHAR(100)
);

-- Estado operativo del usuario (activo, suspendido, bloqueado).
-- Separado porque cambia con frecuencia y necesita trazabilidad:
-- se guarda quién hizo el cambio, cuándo, y por qué.
-- cambiado_por referencia a otro usuario (el admin que actuó).
CREATE TABLE estado_usuario (
    estado_id     SERIAL PRIMARY KEY,
    usuario_id    INTEGER NOT NULL UNIQUE
        REFERENCES usuarios(usuario_id) ON DELETE CASCADE,
    estado        VARCHAR(20) DEFAULT 'Activo'
        CHECK (estado IN ('Activo','Suspendido','Bloqueado')),
    motivo_cambio VARCHAR(255),
    cambiado_por  INTEGER REFERENCES usuarios(usuario_id),
    fecha_cambio  TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    ultimo_acceso TIMESTAMPTZ
);

-- Credenciales locales: contraseña hasheada y sal criptográfica.
-- Solo existe para usuarios con tipo_autenticacion = 'Local'.
-- Un usuario OAuth2 o LDAP NO tiene fila aquí.
-- El salt es un valor aleatorio que se mezcla con la contraseña
-- antes de hacer el hash, para que dos usuarios con la misma
-- contraseña tengan hashes diferentes.
CREATE TABLE credenciales_locales (
    credencial_id      SERIAL PRIMARY KEY,
    usuario_id         INTEGER NOT NULL UNIQUE
        REFERENCES usuarios(usuario_id) ON DELETE CASCADE,
    salt               VARCHAR(100) NOT NULL DEFAULT gen_random_uuid()::text,
    contrasena_hash    VARCHAR(255) NOT NULL,
    ultima_cambio_pass TIMESTAMPTZ  DEFAULT CURRENT_TIMESTAMP,
    politica_id        INTEGER DEFAULT 1
        REFERENCES politicas_contrasenas(politica_id)
);

-- Estado del proceso de autenticación: cuántos intentos fallidos
-- lleva el usuario y hasta cuándo está bloqueado.
-- Separado de credenciales porque se actualiza en CADA intento
-- de login (alta frecuencia), mientras que la contraseña cambia poco.
-- Esto reduce la contención de bloqueos en la BD bajo ataques.
CREATE TABLE estado_credencial (
    estado_cred_id    SERIAL PRIMARY KEY,
    usuario_id        INTEGER NOT NULL UNIQUE
        REFERENCES usuarios(usuario_id) ON DELETE CASCADE,
    intentos_fallidos INTEGER     DEFAULT 0,
    bloqueado_hasta   TIMESTAMPTZ,
    ultimo_intento    TIMESTAMPTZ
);

-- Tabla puente: asigna uno o más roles a cada usuario.
CREATE TABLE usuario_rol (
    usuario_id       INTEGER NOT NULL REFERENCES usuarios(usuario_id) ON DELETE CASCADE,
    rol_id           INTEGER NOT NULL REFERENCES roles(rol_id)        ON DELETE CASCADE,
    fecha_asignacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (usuario_id, rol_id)
);


-- =====================================================
-- BLOQUE 2-B: RESET DE CONTRASEÑA  (RF1.3 — GAP 1)
-- =====================================================

-- Cubre el requerimiento RF1.3 "Recuperar contraseña" del SRS,
-- que no tenía soporte en la BD original.
--
-- Flujo típico:
--   1. El usuario solicita recuperar su contraseña (olvidé mi clave).
--   2. El backend genera un token aleatorio de un solo uso,
--      lo hashea (igual que con los tokens de sesión) y lo guarda aquí.
--   3. Se envía el token en claro por email al usuario.
--   4. El usuario hace clic en el enlace → el backend busca el hash,
--      verifica que no esté usado ni expirado, y permite el cambio.
--   5. Se marca usado = TRUE para que el enlace no pueda reutilizarse.
--
-- Diferencia clave con la tabla 'tokens':
--   · Los tokens de sesión (access/refresh) viven mientras la sesión dura.
--   · El token de reset caduca en minutos (15–30 min), es de un solo uso
--     y no crea sesión: solo autoriza el cambio de contraseña.
--   · Separarlos evita confusión de lógica en el backend y
--     permite políticas de expiración distintas.
--
-- token_hash: se hashea con SHA-256 vía pgcrypto igual que los demás tokens.
-- expira_en:  el backend debe rechazar tokens cuya fecha ya pasó.
-- usado:      TRUE en cuanto se completa el reset; previene reuso del enlace.
-- ip_solicitud: IP desde donde se pidió el reset (auditoría y detección de abuso).
CREATE TABLE reset_contrasena (
    reset_id       SERIAL PRIMARY KEY,
    usuario_id     INTEGER      NOT NULL
        REFERENCES usuarios(usuario_id) ON DELETE CASCADE,
    token_hash     VARCHAR(500) NOT NULL UNIQUE,  -- hash del token enviado por email
    expira_en      TIMESTAMPTZ  NOT NULL,          -- corta vida: 15-30 min recomendado
    usado          BOOLEAN      DEFAULT FALSE,     -- TRUE tras completar el reset
    ip_solicitud   VARCHAR(50),                    -- IP que pidió el reset
    creado_en      TIMESTAMPTZ  DEFAULT CURRENT_TIMESTAMP,
    usado_en       TIMESTAMPTZ                     -- cuándo se usó (para auditoría)
);

-- Índice para que el backend encuentre rápido tokens activos (no usados, no expirados).
-- Es parcial: solo indexa los tokens donde usado = FALSE, ignorando los ya consumidos.
-- Más rápido y ocupa menos espacio que un índice sobre toda la tabla.
CREATE INDEX idx_reset_usuario_activo
    ON reset_contrasena(usuario_id, expira_en)
    WHERE usado = FALSE;


-- =====================================================
-- BLOQUE 2-C: DOBLE FACTOR DE AUTENTICACIÓN — 2FA  (GAP 2)
-- =====================================================

-- El SRS (RNF5) pide autenticación segura. Para un sistema IoT
-- con datos domésticos, el 2FA es la capa adicional recomendada:
-- incluso si alguien roba la contraseña, no puede entrar sin el
-- segundo factor (código temporal o llave física).
--
-- Esta tabla guarda la configuración 2FA de cada usuario.
-- No todos los usuarios tienen 2FA activo: solo quienes lo habiliten.
--
-- metodo:
--   'TOTP'     → app de autenticador (Google Authenticator, Authy…).
--                Genera un código de 6 dígitos cada 30 segundos usando
--                el secreto guardado aquí. Es el más común y seguro.
--   'SMS'      → código de un solo uso enviado al teléfono del usuario.
--                Menos seguro que TOTP (vulnerable a SIM-swap) pero
--                más fácil de usar para usuarios no técnicos.
--   'Email'    → código enviado al correo. Más lento pero sin app extra.
--
-- secreto_totp: clave secreta compartida entre el servidor y la app
--   del autenticador. Se cifra en la BD (no en texto claro).
--   Solo se usa cuando metodo = 'TOTP'.
--
-- codigos_respaldo: hasta 10 códigos de un solo uso para cuando
--   el usuario pierde el teléfono. Se guardan hasheados en JSON:
--   [{"hash": "abc123...", "usado": false}, ...]
--   Esto permite verificarlos sin guardarlos en texto claro.
--
-- activo: permite al usuario habilitar/deshabilitar 2FA sin
--   borrar la configuración (no pierde el secreto TOTP).
--
-- verificado: FALSE mientras el usuario no confirma que el setup
--   funciona correctamente. Evita que se quede fuera por configurar
--   mal el autenticador.
CREATE TABLE usuario_2fa (
    twofa_id          SERIAL PRIMARY KEY,
    usuario_id        INTEGER      NOT NULL UNIQUE
        REFERENCES usuarios(usuario_id) ON DELETE CASCADE,
    metodo            VARCHAR(10)  NOT NULL DEFAULT 'TOTP'
        CHECK (metodo IN ('TOTP','SMS','Email')),
    secreto_totp      VARCHAR(255),            -- cifrado; solo para metodo='TOTP'
    codigos_respaldo  JSONB,                   -- array de hashes de códigos de emergencia
    activo            BOOLEAN      DEFAULT FALSE,
    verificado        BOOLEAN      DEFAULT FALSE, -- TRUE cuando el usuario confirmó el setup
    fecha_activacion  TIMESTAMPTZ,               -- cuándo se activó por primera vez
    ultimo_uso        TIMESTAMPTZ               -- útil para detectar cuenta inactiva
);

-- Índice para consultar rápido si un usuario tiene 2FA activo.
-- El backend lo consulta en cada login para saber si debe pedir el segundo factor.
CREATE INDEX idx_2fa_usuario_activo
    ON usuario_2fa(usuario_id)
    WHERE activo = TRUE AND verificado = TRUE;


-- =====================================================
-- BLOQUE 3: DOMINIO — HOGARES Y MEDIDORES
-- =====================================================

-- Un hogar es la unidad de consumo (casa, apartamento, local).
-- estrato va de 1 a 6 según la clasificación colombiana de servicios.
-- NOTA: valor_base fue eliminado; la tarifa ahora tiene historial
-- propio en hogar_tarifa para poder recalcular facturas pasadas.
CREATE TABLE hogares (
    hogar_id   SERIAL PRIMARY KEY,
    nombre     VARCHAR(100) NOT NULL,
    direccion  VARCHAR(255),
    ciudad     VARCHAR(100),
    estrato    SMALLINT CHECK (estrato BETWEEN 1 AND 6),
    titular_id INTEGER NOT NULL REFERENCES usuarios(usuario_id),
    estado     VARCHAR(20) DEFAULT 'Activo'
        CHECK (estado IN ('Activo','Suspendido'))
);

-- Historial de tarifas de cada hogar.
-- vigente_hasta = NULL significa que esa es la tarifa actual.
-- El constraint EXCLUDE evita que dos tarifas se solapen en el tiempo
-- para el mismo hogar (requiere la extensión btree_gist).
-- Ejemplo: si hay una tarifa de enero a marzo, no puedes crear
-- otra que empiece en febrero para el mismo hogar.
CREATE TABLE hogar_tarifa (
    tarifa_id     SERIAL PRIMARY KEY,
    hogar_id      INTEGER NOT NULL REFERENCES hogares(hogar_id) ON DELETE CASCADE,
    valor_m3      DECIMAL(12,4) NOT NULL,   -- precio por metro cúbico
    cargo_fijo    DECIMAL(12,2) DEFAULT 0,  -- cargo fijo mensual (alcantarillado, etc.)
    vigente_desde DATE NOT NULL,
    vigente_hasta DATE,                     -- NULL = tarifa vigente actualmente
    fuente        VARCHAR(100),             -- ej: 'Resolución SSPD 2025-03'
    CONSTRAINT no_solapamiento_tarifa EXCLUDE USING gist (
        hogar_id WITH =,
        daterange(vigente_desde, vigente_hasta, '[)') WITH &&
    )
);

-- Miembros de un hogar: el titular más quienes viven ahí.
-- rol_en_hogar define qué puede hacer cada miembro:
--   'Titular'  → dueño, puede todo
--   'Miembro'  → puede ver consumo y alertas
--   'Invitado' → acceso de solo lectura temporal
-- permisos_especiales: JSON libre para casos edge
-- (ej: un miembro con permiso temporal de editar configuración).
CREATE TABLE hogar_miembro (
    hogar_id            INTEGER NOT NULL REFERENCES hogares(hogar_id)   ON DELETE CASCADE,
    usuario_id          INTEGER NOT NULL REFERENCES usuarios(usuario_id) ON DELETE CASCADE,
    rol_en_hogar        VARCHAR(20) DEFAULT 'Miembro'
        CHECK (rol_en_hogar IN ('Titular','Miembro','Invitado')),
    permisos_especiales JSONB,
    fecha_asignacion    TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (hogar_id, usuario_id)
);

-- Configuración personalizada de cada hogar:
-- consumo_tope (m³ antes de alertar), idioma, moneda, tema visual.
-- notif_* controlan si el hogar quiere notificaciones por cada canal.
-- consumo_tope_diario: límite diario para alertas más granulares.
-- NOTA: alertas_config JSONB fue reemplazado por columnas tipadas
-- para evitar que entre cualquier estructura sin validación.
CREATE TABLE configuraciones (
    config_id            SERIAL PRIMARY KEY,
    hogar_id             INTEGER UNIQUE NOT NULL REFERENCES hogares(hogar_id),
    consumo_tope         DECIMAL(10,2),
    consumo_tope_diario  DECIMAL(10,2),
    notif_app            BOOLEAN DEFAULT TRUE,
    notif_email          BOOLEAN DEFAULT TRUE,
    notif_sms            BOOLEAN DEFAULT FALSE,
    idioma               VARCHAR(10) DEFAULT 'es',
    moneda               VARCHAR(3)  DEFAULT 'COP',
    tema_color           VARCHAR(50) DEFAULT 'default'
);

-- Un medidor es el dispositivo físico (ej: ESP32) que mide el agua.
-- Puede estar asociado a un hogar o ser independiente (usuario sin hogar).
-- NOTA: estado_medidor fue eliminado de aquí; ahora el estado tiene
-- historial completo en medidor_estado_hist.
CREATE TABLE medidores (
    medidor_id        SERIAL PRIMARY KEY,
    hogar_id          INTEGER      REFERENCES hogares(hogar_id),
    usuario_id        INTEGER NOT NULL REFERENCES usuarios(usuario_id),
    codigo_medidor    VARCHAR(100) UNIQUE NOT NULL,
    tipo              VARCHAR(50),
    ubicacion         VARCHAR(255),
    fecha_instalacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- Historial completo de cambios de estado del medidor.
-- fecha_hasta = NULL significa que ese es el estado actual.
-- Permite saber exactamente cuánto tiempo estuvo en mantenimiento,
-- quién hizo el cambio y por qué.
CREATE TABLE medidor_estado_hist (
    hist_id        SERIAL PRIMARY KEY,
    medidor_id     INTEGER NOT NULL REFERENCES medidores(medidor_id) ON DELETE CASCADE,
    estado         VARCHAR(20) NOT NULL
        CHECK (estado IN ('Activo','Suspendido','Mantenimiento')),
    motivo         VARCHAR(255),
    registrado_por INTEGER REFERENCES usuarios(usuario_id),
    fecha_desde    TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_hasta    TIMESTAMPTZ          -- NULL = estado actual
);

-- Vista que muestra el estado actual de cada medidor sin necesidad
-- de consultar todo el historial. DISTINCT ON es una forma eficiente
-- de quedarse solo con la fila más reciente por medidor_id.
CREATE VIEW v_medidor_estado_actual AS
SELECT DISTINCT ON (medidor_id)
    medidor_id, estado, motivo, fecha_desde
FROM medidor_estado_hist
ORDER BY medidor_id, fecha_desde DESC;


-- =====================================================
-- BLOQUE 4: CONSUMO
-- =====================================================

-- Lectura cruda del sensor: el dato exacto que envió el dispositivo.
-- Esta tabla es INMUTABLE una vez insertada; no se edita ni recalcula.
-- actividad: categoría de uso del agua en ese momento.
-- flujo: litros por minuto en ese instante (dato físico del sensor).
CREATE TABLE consumo_lectura (
    lectura_id         SERIAL PRIMARY KEY,
    medidor_id         INTEGER NOT NULL REFERENCES medidores(medidor_id) ON DELETE CASCADE,
    fecha_lectura      TIMESTAMPTZ(3) NOT NULL,   -- precisión de milisegundos
    cantidad_consumida DECIMAL(10,4)  NOT NULL,   -- metros cúbicos
    actividad          VARCHAR(50)    DEFAULT 'General'
        CHECK (actividad IN ('Ducha','Lavado','Cocina','Aseo','Otros','General')),
    flujo              DECIMAL(8,4)               -- litros/minuto
);

-- Métricas calculadas sobre cada lectura: costo, tarifa aplicada.
-- Separadas de la lectura cruda para poder recalcular si cambia
-- el algoritmo de facturación sin perder los datos originales.
-- tarifa_aplicada: snapshot del precio en el momento del cálculo
-- (importante para auditoría; la tarifa puede cambiar después).
CREATE TABLE consumo_metrica (
    metrica_id      SERIAL PRIMARY KEY,
    lectura_id      INTEGER NOT NULL UNIQUE
        REFERENCES consumo_lectura(lectura_id) ON DELETE CASCADE,
    costo_calculado DECIMAL(10,2),
    tarifa_aplicada DECIMAL(10,4),
    calculado_en    TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);


-- =====================================================
-- BLOQUE 4-B: PREDICCIONES DE CONSUMO  (RF5 — GAP 3)
-- =====================================================

-- Cubre el paso 3 del Caso de Uso N°5 del SRS:
-- "Genera predicciones para mejor administración de dispositivos".
--
-- El backend (ML o algoritmo estadístico) calcula predicciones y
-- las guarda aquí para que la app las muestre sin recalcular
-- en cada request, y para poder auditar la calidad del modelo
-- comparando predicción vs consumo real después.
--
-- Diseño de la tabla:
--
-- tipo_modelo: identifica qué algoritmo generó la predicción.
--   'promedio_movil' → simple, sin ML, basado en los últimos N días.
--   'regresion'      → regresión lineal o polinomial sobre histórico.
--   'ml_externo'     → modelo entrenado fuera (Python, TensorFlow, etc.)
--                      y cuyo resultado se importa aquí.
--
-- periodo_inicio / periodo_fin: rango de fechas al que aplica
--   la predicción (ej: "para el mes de abril de 2026").
--
-- cantidad_predicha: consumo estimado en m³ para ese periodo.
--
-- intervalo_confianza_inf / sup: rango estadístico de la predicción.
--   Ej: "entre 12.5 y 18.3 m³ con 95 % de confianza".
--   Permite mostrar en la app no solo el número sino su incertidumbre.
--
-- cantidad_real: se llena después, cuando el periodo ya pasó,
--   con el consumo efectivo. Permite medir el error del modelo
--   (MAE, MAPE) para mejorar predicciones futuras.
--
-- metadatos: JSON libre para parámetros del modelo, versión,
--   features usados, etc. No todos los modelos tienen los mismos
--   parámetros, así que JSONB da flexibilidad sin romper el esquema.
CREATE TABLE prediccion_consumo (
    prediccion_id           SERIAL PRIMARY KEY,
    hogar_id                INTEGER      NOT NULL
        REFERENCES hogares(hogar_id) ON DELETE CASCADE,
    medidor_id              INTEGER
        REFERENCES medidores(medidor_id) ON DELETE SET NULL,
    tipo_modelo             VARCHAR(50)  NOT NULL DEFAULT 'promedio_movil'
        CHECK (tipo_modelo IN ('promedio_movil','regresion','ml_externo')),
    periodo_inicio          DATE         NOT NULL,  -- inicio del periodo predicho
    periodo_fin             DATE         NOT NULL,  -- fin del periodo predicho
    cantidad_predicha       DECIMAL(10,4) NOT NULL, -- m³ estimados
    intervalo_confianza_inf DECIMAL(10,4),           -- límite inferior del IC
    intervalo_confianza_sup DECIMAL(10,4),           -- límite superior del IC
    cantidad_real           DECIMAL(10,4),           -- se llena al cerrar el periodo
    generado_en             TIMESTAMPTZ  DEFAULT CURRENT_TIMESTAMP,
    metadatos               JSONB,                  -- parámetros y versión del modelo
    CONSTRAINT periodo_valido CHECK (periodo_fin > periodo_inicio),
    CONSTRAINT confianza_coherente CHECK (
        intervalo_confianza_inf IS NULL OR
        intervalo_confianza_inf <= cantidad_predicha
    ),
    CONSTRAINT confianza_coherente_sup CHECK (
        intervalo_confianza_sup IS NULL OR
        intervalo_confianza_sup >= cantidad_predicha
    )
);

-- Índice para que la app encuentre rápido las predicciones
-- vigentes o futuras de un hogar (lo más consultado en el dashboard).
CREATE INDEX idx_prediccion_hogar_periodo
    ON prediccion_consumo(hogar_id, periodo_inicio DESC);

-- Índice para comparar predicción vs real una vez que el periodo cierra.
-- Útil para el módulo de métricas de calidad del modelo.
CREATE INDEX idx_prediccion_sin_real
    ON prediccion_consumo(hogar_id, periodo_fin)
    WHERE cantidad_real IS NULL;


-- =====================================================
-- BLOQUE 5: ALERTAS
-- =====================================================

-- Definición de una regla de alerta: quién la configuró,
-- qué condición la dispara y por qué canal notificar.
-- umbral: valor límite (ej: 15 m³/mes, 50 L/día).
-- activa: permite pausar una regla sin eliminarla.
CREATE TABLE reglas_alerta (
    regla_id       SERIAL PRIMARY KEY,
    hogar_id       INTEGER REFERENCES hogares(hogar_id),
    medidor_id     INTEGER REFERENCES medidores(medidor_id),
    usuario_id     INTEGER NOT NULL REFERENCES usuarios(usuario_id),
    tipo_alerta    VARCHAR(100) NOT NULL,
    umbral         DECIMAL(10,4),
    canal_notif    VARCHAR(20) DEFAULT 'App'
        CHECK (canal_notif IN ('App','Email','SMS')),
    activa         BOOLEAN DEFAULT TRUE,
    fecha_creacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- Evento concreto generado cuando una regla se dispara.
-- Separado de la regla para preservar el historial aunque
-- la regla sea modificada o eliminada posteriormente.
-- estado del evento:
--   'Pendiente'  → generada, aún no enviada
--   'Enviada'    → notificación enviada al usuario
--   'Leida'      → el usuario la vio en la app
--   'Descartada' → el usuario la ignoró explícitamente
CREATE TABLE eventos_alerta (
    evento_id        SERIAL PRIMARY KEY,
    regla_id         INTEGER NOT NULL REFERENCES reglas_alerta(regla_id),
    usuario_id       INTEGER NOT NULL REFERENCES usuarios(usuario_id),
    mensaje          VARCHAR(500),
    fecha_generacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    estado           VARCHAR(20) DEFAULT 'Pendiente'
        CHECK (estado IN ('Pendiente','Enviada','Leida','Descartada'))
);


-- =====================================================
-- BLOQUE 6: SESIONES Y TOKENS
-- =====================================================

-- Un token es una credencial temporal que prueba que el usuario
-- ya se autenticó. Hay dos tipos:
--   'access'  → se envía en cada petición HTTP; caduca rápido (15 min)
--   'refresh' → se usa solo para pedir un nuevo access token; caduca lento (7 días)
-- token_hash: se guarda el HASH del token, nunca el token en claro
-- (igual que con contraseñas; si roban la BD no pueden usarlos).
-- parent_id: el refresh token que originó este access token.
-- Revocar el refresh token cascadea a todos sus access tokens hijos.
CREATE TABLE tokens (
    token_id   SERIAL PRIMARY KEY,
    usuario_id INTEGER NOT NULL REFERENCES usuarios(usuario_id) ON DELETE CASCADE,
    tipo       VARCHAR(10) NOT NULL
        CHECK (tipo IN ('access','refresh')),
    token_hash VARCHAR(500) UNIQUE NOT NULL,
    parent_id  INTEGER REFERENCES tokens(token_id),  -- refresh → access
    expira_en  TIMESTAMPTZ NOT NULL,
    estado     VARCHAR(20) DEFAULT 'Activo'
        CHECK (estado IN ('Activo','Revocado','Expirado')),
    ip_origen  VARCHAR(50),
    user_agent VARCHAR(500),
    creado_en  TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- Registro de cada sesión de usuario: desde qué IP y dispositivo
-- entró, cuándo empezó y cuándo terminó.
-- token_id: FK real al token activo de esa sesión.
-- ON DELETE SET NULL: si el token se borra, la sesión queda
-- con token_id = NULL (sesión huérfana) sin eliminar el historial.
CREATE TABLE sesion_usuario (
    sesion_id     SERIAL PRIMARY KEY,
    usuario_id    INTEGER NOT NULL REFERENCES usuarios(usuario_id),
    token_id      INTEGER REFERENCES tokens(token_id) ON DELETE SET NULL,
    fecha_inicio  TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    fecha_fin     TIMESTAMPTZ,
    estado_sesion VARCHAR(20)
        CHECK (estado_sesion IN ('Activo','Cerrado','Expirado')),
    ip_origen     VARCHAR(50),
    dispositivo   VARCHAR(200),
    -- campo nuevo: indica si esta sesión pasó verificación 2FA.
    -- NULL  → el usuario no tiene 2FA activo (no aplica).
    -- FALSE → llegó hasta aquí pero aún no completó el segundo factor.
    -- TRUE  → segundo factor verificado; sesión completamente autenticada.
    -- El backend debe bloquear el acceso a recursos protegidos si
    -- twofa_verificado = FALSE cuando el usuario tiene 2FA activo.
    twofa_verificado BOOLEAN DEFAULT NULL
);


-- =====================================================
-- BLOQUE 7: AUDITORÍA Y LOGS
-- =====================================================

-- Bitácora de todas las acciones importantes del sistema.
-- Se llena automáticamente mediante triggers (ver 04_triggers.sql).
-- PARTITION BY RANGE (fecha): divide la tabla en sub-tablas mensuales.
-- Ventaja: borrar un mes de auditoría es instantáneo (DROP TABLE
-- sobre la partición), en lugar de un DELETE lento sobre millones de filas.
-- Las consultas por rango de fechas también son mucho más rápidas
-- porque PostgreSQL sabe en qué partición buscar.
CREATE TABLE auditoria (
    auditoria_id   BIGSERIAL,           -- BIGSERIAL porque puede crecer mucho
    usuario_id     INTEGER REFERENCES usuarios(usuario_id),
    accion         VARCHAR(255),
    tabla_afectada VARCHAR(100),
    registro_id    INTEGER,
    fecha          TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    descripcion    VARCHAR(500),
    ip_origen      VARCHAR(50),
    aplicacion     VARCHAR(255),
    user_agent     VARCHAR(500)
) PARTITION BY RANGE (fecha);

-- Particiones mensuales de auditoría (agregar una nueva cada mes).
-- El rango es [desde, hasta): incluye el primer día, excluye el último.
CREATE TABLE auditoria_2026_01 PARTITION OF auditoria
    FOR VALUES FROM ('2026-01-01') TO ('2026-02-01');
CREATE TABLE auditoria_2026_02 PARTITION OF auditoria
    FOR VALUES FROM ('2026-02-01') TO ('2026-03-01');
CREATE TABLE auditoria_2026_03 PARTITION OF auditoria
    FOR VALUES FROM ('2026-03-01') TO ('2026-04-01');
CREATE TABLE auditoria_2026_04 PARTITION OF auditoria
    FOR VALUES FROM ('2026-04-01') TO ('2026-05-01');
CREATE TABLE auditoria_2026_05 PARTITION OF auditoria
    FOR VALUES FROM ('2026-05-01') TO ('2026-06-01');
CREATE TABLE auditoria_2026_06 PARTITION OF auditoria
    FOR VALUES FROM ('2026-06-01') TO ('2026-07-01');
CREATE TABLE auditoria_2026_07 PARTITION OF auditoria
    FOR VALUES FROM ('2026-07-01') TO ('2026-08-01');
CREATE TABLE auditoria_2026_08 PARTITION OF auditoria
    FOR VALUES FROM ('2026-08-01') TO ('2026-09-01');
CREATE TABLE auditoria_2026_09 PARTITION OF auditoria
    FOR VALUES FROM ('2026-09-01') TO ('2026-10-01');
CREATE TABLE auditoria_2026_10 PARTITION OF auditoria
    FOR VALUES FROM ('2026-10-01') TO ('2026-11-01');
CREATE TABLE auditoria_2026_11 PARTITION OF auditoria
    FOR VALUES FROM ('2026-11-01') TO ('2026-12-01');
CREATE TABLE auditoria_2026_12 PARTITION OF auditoria
    FOR VALUES FROM ('2026-12-01') TO ('2027-01-01');

-- Errores del sistema: excepciones del backend, fallos de login, etc.
-- severidad:
--   'INFO'     → informativo, sin problema real
--   'WARN'     → algo raro pero el sistema siguió funcionando
--   'ERROR'    → algo falló, el usuario recibió un error
--   'CRITICAL' → el sistema está en riesgo, requiere atención inmediata
CREATE TABLE log_errores (
    error_id    SERIAL PRIMARY KEY,
    usuario_id  INTEGER REFERENCES usuarios(usuario_id),
    fecha       TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    tipo_error  VARCHAR(100),
    descripcion VARCHAR(1000),
    ip_origen   VARCHAR(50),
    stack_trace TEXT,        -- rastro completo del error (para desarrolladores)
    severidad   VARCHAR(20) DEFAULT 'ERROR'
        CHECK (severidad IN ('INFO','WARN','ERROR','CRITICAL'))
);


-- =====================================================
-- ÍNDICES — TABLAS ORIGINALES
-- =====================================================

-- Un índice acelera las búsquedas sobre una columna,
-- igual que el índice de un libro: en lugar de leer
-- todas las páginas, vas directo al resultado.
-- Costo: ocupan espacio en disco y ralentizan los INSERT/UPDATE.
-- Por eso solo se crean en columnas que se usan frecuentemente
-- en filtros (WHERE) o en joins.

-- Consumo: búsquedas por medidor y fecha (las más comunes en reportes)
CREATE INDEX idx_lectura_medidor_fecha  ON consumo_lectura(medidor_id, fecha_lectura);
CREATE INDEX idx_lectura_actividad      ON consumo_lectura(actividad);

-- Alertas
CREATE INDEX idx_eventos_alerta_usuario ON eventos_alerta(usuario_id);
CREATE INDEX idx_eventos_alerta_regla   ON eventos_alerta(regla_id, estado);
CREATE INDEX idx_reglas_alerta_hogar    ON reglas_alerta(hogar_id);

-- Medidores
CREATE INDEX idx_medidores_hogar        ON medidores(hogar_id);
CREATE INDEX idx_medidor_estado_actual  ON medidor_estado_hist(medidor_id, fecha_desde DESC);

-- Auditoría
CREATE INDEX idx_auditoria_usuario      ON auditoria(usuario_id, fecha);

-- Tokens: índice parcial — solo busca entre los tokens activos,
-- ignorando los ya expirados o revocados. Más rápido y pequeño.
CREATE INDEX idx_tokens_usuario_tipo    ON tokens(usuario_id, tipo, estado);
CREATE INDEX idx_tokens_activos         ON tokens(expira_en) WHERE estado = 'Activo';

-- Sesiones
CREATE INDEX idx_sesion_usuario_estado  ON sesion_usuario(usuario_id, estado_sesion);

-- Credenciales
CREATE INDEX idx_credenciales_usuario   ON credenciales_locales(usuario_id);
CREATE INDEX idx_estado_cred_usuario    ON estado_credencial(usuario_id);

-- Hogar → miembro
CREATE INDEX idx_hogar_miembro_usuario  ON hogar_miembro(usuario_id);

-- Tarifas: buscar la tarifa vigente de un hogar rápidamente
CREATE INDEX idx_tarifa_hogar_vigente   ON hogar_tarifa(hogar_id, vigente_desde DESC);

-- Log de errores: filtrar por fecha y severidad
CREATE INDEX idx_log_errores_fecha      ON log_errores(fecha, severidad);

-- Configuración de seguridad: búsqueda por nombre de parámetro
CREATE INDEX idx_cfg_seg_nombre         ON configuracion_seguridad(nombre_configuracion);
