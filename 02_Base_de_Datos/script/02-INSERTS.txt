-- =====================================================
-- BLOQUE 2: ROLES Y PERMISOS
-- =====================================================

-- Los tres roles del sistema.
-- rol_id 1 = Admin, 2 = Soporte Técnico, 3 = Usuario doméstico
-- (el orden de inserción determina el ID con SERIAL)
INSERT INTO roles (nombre_rol, descripcion) VALUES
    ('Admin',             'Administrador total del sistema'),
    ('Soporte Técnico',   'Mantenimiento de dispositivos e infraestructura'),
    ('Usuario Doméstico', 'Usuario final que monitorea su hogar');

-- Permisos disponibles en el sistema.
-- 'funcional': acción de negocio (ver reportes, configurar alertas…)
-- 'dato':      operación SQL directa sobre tablas (SELECT, INSERT…)
INSERT INTO permisos (nombre_permiso, tipo_permiso, descripcion) VALUES
    ('ver_reportes',           'funcional', 'Consultar reportes y gráficos de consumo'),
    ('gestionar_dispositivos', 'funcional', 'Registrar y modificar medidores'),
    ('admin_usuarios',         'funcional', 'Crear, suspender y reactivar usuarios'),
    ('configurar_alertas',     'funcional', 'Definir topes y reglas de notificación'),
    ('ver_auditoria',          'funcional', 'Consultar bitácora de acciones del sistema'),
    ('gestionar_tarifas',      'funcional', 'Crear y modificar historial de tarifas'),
    ('SELECT',                 'dato',      'Permiso de lectura sobre tablas asignadas'),
    ('INSERT',                 'dato',      'Permiso de inserción sobre tablas asignadas'),
    ('UPDATE',                 'dato',      'Permiso de actualización sobre tablas asignadas'),
    ('DELETE',                 'dato',      'Permiso de eliminación sobre tablas asignadas');

-- ── Asignación de permisos por rol ───────────────────

-- ADMIN: acceso total — todos los permisos sin excepción.
INSERT INTO rol_permiso (rol_id, permiso_id)
    SELECT 1, permiso_id FROM permisos;

-- SOPORTE TÉCNICO: puede ver y actualizar dispositivos y reportes,
-- pero NO puede administrar usuarios ni ver auditoría completa.
INSERT INTO rol_permiso (rol_id, permiso_id)
    SELECT 2, permiso_id FROM permisos
    WHERE nombre_permiso IN (
        'gestionar_dispositivos',
        'ver_reportes',
        'SELECT',
        'UPDATE'
    );

-- USUARIO DOMÉSTICO: puede ver su consumo, configurar sus alertas
-- e insertar lecturas. No puede gestionar dispositivos ni usuarios.
INSERT INTO rol_permiso (rol_id, permiso_id)
    SELECT 3, permiso_id FROM permisos
    WHERE nombre_permiso IN (
        'ver_reportes',
        'configurar_alertas',
        'SELECT',
        'INSERT'
    );
